const functions = require("firebase-functions"); const admin = require("firebase-admin"); admin.initializeApp(); async function enviarNotificacion(tokensPath, titulo, cuerpo) { const tokensSnap = await admin.database().ref(tokensPath).once("value"); const tokensObj = tokensSnap.val() || {}; const tokens = Object.keys(tokensObj); if (tokens.length === 0) { console.log(`No hay dispositivos registrados en ${tokensPath} para notificar.`); return; } const mensaje = { notification: { title: titulo, body: cuerpo }, android: { priority: "high", }, webpush: { headers: { Urgency: "high", }, }, tokens: tokens, }; const respuesta = await admin.messaging().sendEachForMulticast(mensaje); console.log(`${tokensPath}: ${respuesta.successCount} exitosas, ${respuesta.failureCount} fallidas.`); const tokensInvalidos = []; respuesta.responses.forEach((r, i) => { if (!r.success) tokensInvalidos.push(tokens[i]); }); if (tokensInvalidos.length > 0) { const actualizaciones = {}; tokensInvalidos.forEach((t) => (actualizaciones[tokensPath + "/" + t] = null)); await admin.database().ref().update(actualizaciones); } } exports.notificarAlerta = functions.database .ref("/aulas") .onWrite(async (change) => { const antes = change.before.val() || []; const despues = change.after.val() || []; const nuevaAlerta = despues.find((aula, i) => { const anterior = antes[i]; return aula && aula.activa && (!anterior || !anterior.activa); }); if (nuevaAlerta) { await enviarNotificacion( "/tokensAuxiliar", "Alerta Aula " + nuevaAlerta.id, "Se necesita auxiliar en el Aula " + nuevaAlerta.id ); return null; } const alertaAtendida = despues.find((aula, i) => { const anterior = antes[i]; return anterior && anterior.activa && aula && !aula.activa; }); if (alertaAtendida) { await enviarNotificacion( "/tokensMaestro", "Aula " + alertaAtendida.id + " atendida", "El auxiliar ya atendio la alerta del Aula " + alertaAtendida.id ); return null; } return null; });
